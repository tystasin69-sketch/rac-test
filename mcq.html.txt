    // Bengali digits translation helper
    function toBengaliNumber(num) {
      const bnDigits = ['০', '১', '২', '৩', '৪', '৫', '৬', '৭', '৮', '৯'];
      return num.toString().replace(/\d/g, d => bnDigits[parseInt(d, 10)]);
    }

    function sendDataToGoogleSheet(payload) {
      if (!SHEET_WEBAPP_URL) return;
      const banner = document.getElementById('syncBanner');
      const text = document.getElementById('syncText');
      banner.classList.remove('hidden');
      text.textContent = "📡 গুগল শিটে ডাটা পাঠানো হচ্ছে...";

      fetch(SHEET_WEBAPP_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      }).then(() => {
        text.textContent = "✓ গুগল শিটে লাইভ ডাটা সংরক্ষিত হয়েছে!";
        setTimeout(() => banner.classList.add('hidden'), 3500);
      }).catch(err => {
        text.textContent = "⚠️ শিটে সিঙ্ক হতে পারেনি (ইন্টারনেট চেক করুন)";
        setTimeout(() => banner.classList.add('hidden'), 3500);
      });
    }

    function checkStudentProfile() {
      if (!currentUser) {
        document.getElementById('studentModal').classList.remove('hidden');
      } else {
        document.getElementById('studentModal').classList.add('hidden');
        document.getElementById('studentPill').classList.remove('hidden');

      }
    }

    function handleStartExamWithStudent(e) {
      e.preventDefault();
      const name = document.getElementById('studentNameInput').value.trim();
      const info = document.getElementById('studentInfoInput').value.trim();

      if (!name || !info) return;

      currentUser = { name, info };
      localStorage.setItem('rac_current_user', JSON.stringify(currentUser));
      document.getElementById('studentModal').classList.add('hidden');
      document.getElementById('studentPill').classList.remove('hidden');
      document.getElementById('studentPillName').textContent = currentUser.name;

      examStartTime = new Date().toLocaleString('bn-BD', { timeZone: 'Asia/Dhaka' });

      // Send Started Log to Google Sheet
      sendDataToGoogleSheet({
        time: new Date().toLocaleString(),
        name: currentUser.name,
        roll: currentUser.info,
        status: 'পরীক্ষা শুরু করেছে',
        score: '০ / ৭৩',
        percentage: '০%'
      });
    }

    function handleChangeUser() {
      currentUser = null;
      localStorage.removeItem('rac_current_user');
      document.getElementById('studentNameInput').value = '';
      document.getElementById('studentInfoInput').value = '';
      document.getElementById('studentPill').classList.add('hidden');
      checkStudentProfile();
    }

    function submitFinalExam() {
      let correct = 0;
      let wrong = 0;
      questionsData.forEach(q => {
        const a = userAnswers[q.id];
        if (a !== undefined) {
          if (a === q.correct) correct++;
          else wrong++;
        }
      });

      const total = questionsData.length;
      const pct = Math.round((correct / total) * 100);

      document.getElementById('resName').textContent = currentUser ? currentUser.name : 'পরীক্ষার্থী';
      document.getElementById('resCorrect').textContent = `${toBengaliNumber(correct)} টি`;
      document.getElementById('resWrong').textContent = `${toBengaliNumber(wrong)} টি`;
      document.getElementById('resPercent').textContent = `${toBengaliNumber(pct)}%`;

      document.getElementById('resultModal').classList.remove('hidden');

      // Send Finished Log to Google Sheet
      if (currentUser) {
        sendDataToGoogleSheet({
          time: new Date().toLocaleString(),
          name: currentUser.name,
          roll: currentUser.info,
          status: 'পরীক্ষা সম্পন্ন (Finished)',
          score: `${correct} / ${total}`,
          percentage: `${pct}%`
        });
      }
    }

    function getValidAttempts() {
      const raw = localStorage.getItem('rac_attempts_timestamps');
      if (!raw) return [];
      const now = Date.now();
      try {
        const timestamps = JSON.parse(raw);
        const valid = timestamps.filter(ts => (now - ts) < TWENTY_FOUR_HOURS);
        if (valid.length !== timestamps.length) {
          localStorage.setItem('rac_attempts_timestamps', JSON.stringify(valid));
        }
        return valid;
      } catch(e) {
        return [];
      }
    }

    function recordNewAttempt() {
      if (isAdmin) return; // Admin is unlimited
      const attempts = getValidAttempts();
      attempts.push(Date.now());
      localStorage.setItem('rac_attempts_timestamps', JSON.stringify(attempts));
      updateAttemptUI();
    }

    function updateAttemptUI() {
      const badge = document.getElementById('attemptBadge');
      const adminTxt = document.getElementById('adminStatusText');
      const adminIcon = document.getElementById('adminLockIcon');

      if (isAdmin) {
        badge.className = "text-xs font-bold px-2.5 py-1 rounded-lg border bg-amber-50 text-amber-800 border-amber-300 flex items-center gap-1 shadow-sm";
        badge.innerHTML = "👑 অ্যাডমিন মুড অন (আনলিমিটেড পরীক্ষা)";
        adminTxt.textContent = "অ্যাডমিন (Active)";
        adminIcon.textContent = "👑";
        return;
      }

      adminTxt.textContent = "অ্যাডমিন এক্সেস ও শিট সেটআপ";
      adminIcon.textContent = "🔒";

      const attempts = getValidAttempts();
      const left = Math.max(0, MAX_DAILY_ATTEMPTS - attempts.length);

      if (left > 0) {
        badge.className = "text-xs font-bold px-2.5 py-1 rounded-lg border bg-emerald-50 text-emerald-800 border-emerald-300 flex items-center gap-1 shadow-sm";
        badge.innerHTML = `🎯 সুযোগ বাকি: ${toBengaliNumber(left)}/${toBengaliNumber(MAX_DAILY_ATTEMPTS)}`;
      } else {
        badge.className = "text-xs font-bold px-2.5 py-1 rounded-lg border bg-rose-50 text-rose-800 border-rose-300 flex items-center gap-1 shadow-sm";
        badge.innerHTML = `⚠️ আজকের সুযোগ শেষ (${toBengaliNumber(MAX_DAILY_ATTEMPTS)}/${toBengaliNumber(MAX_DAILY_ATTEMPTS)})`;
      }
    }

    function canTakeExam() {
      if (isAdmin) return true;
      const attempts = getValidAttempts();
      return attempts.length < MAX_DAILY_ATTEMPTS;
    }

    function showCountdownForNextAttempt() {
      if (countdownInterval) clearInterval(countdownInterval);

      const updateTimer = () => {
        const attempts = getValidAttempts();
        if (attempts.length === 0) {
          closeLimitModal();
          updateAttemptUI();
          return;
        }
        const oldestAttempt = Math.min(...attempts);
        const nextAvailableTime = oldestAttempt + TWENTY_FOUR_HOURS;
        const remainingMs = nextAvailableTime - Date.now();

        if (remainingMs <= 0) {
          clearInterval(countdownInterval);
          updateAttemptUI();
          closeLimitModal();
          return;
        }

        const hours = Math.floor(remainingMs / (1000 * 60 * 60));
        const mins = Math.floor((remainingMs % (1000 * 60 * 60)) / (1000 * 60));
        const secs = Math.floor((remainingMs % (1000 * 60)) / 1000);

        const pad = (n) => n.toString().padStart(2, '0');
        const timerEl = document.getElementById('countdownTimer');
        if (timerEl) {
          timerEl.textContent = `${toBengaliNumber(pad(hours))}:${toBengaliNumber(pad(mins))}:${toBengaliNumber(pad(secs))}`;
        }
      };

      updateTimer();
      countdownInterval = setInterval(updateTimer, 1000);
    }

    function renderQuiz() {
      const container = document.getElementById('questionsContainer');
      container.innerHTML = '';

      const prefixes = ['ক', 'খ', 'গ', 'ঘ'];
      let visibleCount = 0;

      questionsData.forEach((item) => {
        const isAnswered = userAnswers[item.id] !== undefined;
        const selectedOpt = userAnswers[item.id];
        const isCorrect = isAnswered && selectedOpt === item.correct;

        // Filtering check
        if (currentFilter === 'unanswered' && isAnswered) return;
        if (currentFilter === 'correct' && (!isAnswered || !isCorrect)) return;
        if (currentFilter === 'incorrect' && (!isAnswered || isCorrect)) return;

        visibleCount++;

        const card = document.createElement('div');
        card.id = `q-card-${item.id}`;
        card.className = "bg-white border border-slate-200 hover:border-slate-300 rounded-2xl p-4 sm:p-5 transition shadow-sm";

        // Question Title and Status Badge
        const header = document.createElement('div');
        header.className = "flex items-start justify-between gap-3 mb-3.5";

        const qText = document.createElement('h3');
        qText.className = "text-base sm:text-lg font-bold text-slate-900 leading-relaxed";
        qText.textContent = `${toBengaliNumber(item.id)}। ${item.question}`;
        header.appendChild(qText);

        if (isAnswered) {
          const badge = document.createElement('span');
          badge.className = isCorrect 
            ? "inline-flex items-center gap-1 text-xs font-bold text-emerald-800 bg-emerald-100 border border-emerald-300 px-2.5 py-0.5 rounded-full whitespace-nowrap shadow-xs"
            : "inline-flex items-center gap-1 text-xs font-bold text-rose-800 bg-rose-100 border border-rose-300 px-2.5 py-0.5 rounded-full whitespace-nowrap shadow-xs";
          badge.textContent = isCorrect ? "✓ সঠিক উত্তর" : "✗ ভুল উত্তর";
          header.appendChild(badge);
        }

        card.appendChild(header);

        // Options Layout Grid
        const optionsGrid = document.createElement('div');
        optionsGrid.className = "grid grid-cols-1 sm:grid-cols-2 gap-2.5 mt-2";

        item.options.forEach((optText, optIdx) => {
          const btn = document.createElement('button');
          btn.type = "button";
          btn.disabled = isAnswered;

          let btnClasses = "w-full text-left p-3.5 rounded-xl border text-sm sm:text-base font-semibold transition flex items-start gap-3 ";

          if (!isAnswered) {
            btnClasses += "border-slate-200 bg-slate-50 hover:bg-sky-50 hover:border-sky-300 text-slate-800 cursor-pointer active:scale-[0.99]";
          } else {
            btnClasses += "cursor-default ";
            if (optIdx === item.correct) {
              btnClasses += "border-emerald-500 bg-emerald-50 text-emerald-900 shadow-sm ring-1 ring-emerald-500 font-bold";
            } else if (optIdx === selectedOpt && !isCorrect) {
              btnClasses += "border-rose-500 bg-rose-50 text-rose-900 line-through ring-1 ring-rose-500 font-bold";
            } else {
              btnClasses += "border-slate-200 bg-slate-100 text-slate-400 opacity-60";
            }
          }

          btn.className = btnClasses;

          // Prefix (ক, খ, গ, ঘ)
          const prefixSpan = document.createElement('span');
          let prefixClasses = "w-6 h-6 rounded-full flex items-center justify-center text-xs font-bold shrink-0 mt-0.5 ";

          if (isAnswered && optIdx === item.correct) {
            prefixClasses += "bg-emerald-600 text-white";
          } else if (isAnswered && optIdx === selectedOpt && !isCorrect) {
            prefixClasses += "bg-rose-600 text-white";
          } else {
            prefixClasses += "bg-slate-200 text-slate-700";
          }
          prefixSpan.className = prefixClasses;
          prefixSpan.textContent = prefixes[optIdx] || (optIdx + 1);

          const textSpan = document.createElement('span');
          textSpan.className = "flex-1";
          textSpan.textContent = optText;

          btn.appendChild(prefixSpan);
          btn.appendChild(textSpan);

          btn.onclick = () => handleAnswerSelect(item.id, optIdx);
          optionsGrid.appendChild(btn);
        });

        card.appendChild(optionsGrid);
        container.appendChild(card);
      });

      const emptyNotice = document.getElementById('emptyNotice');
      if (visibleCount === 0) {
        emptyNotice.classList.remove('hidden');
      } else {
        emptyNotice.classList.add('hidden');
      }

      updateLiveStats();
      renderPalette();
    }

    function handleAnswerSelect(questionId, optionIndex) {
      if (userAnswers[questionId] !== undefined) return;
      
      // Check attempt permission on first answer
      if (Object.keys(userAnswers).length === 0) {
        if (!canTakeExam()) {
          showLimitModal();
          return;
        }
      }

      userAnswers[questionId] = optionIndex;
      renderQuiz();
    }

    function updateLiveStats() {
      let correctCount = 0;
      let wrongCount = 0;
      const total = questionsData.length;

      questionsData.forEach(q => {
        const ans = userAnswers[q.id];
        if (ans !== undefined) {
          if (ans === q.correct) {
            correctCount++;
          } else {
            wrongCount++;
          }
        }
      });

      const answeredCount = correctCount + wrongCount;
      const remainingCount = total - answeredCount;

      document.getElementById('scoreText').textContent = toBengaliNumber(correctCount);
      document.getElementById('answeredText').textContent = toBengaliNumber(answeredCount);

      document.getElementById('countUnanswered').textContent = toBengaliNumber(remainingCount);
      document.getElementById('countCorrect').textContent = toBengaliNumber(correctCount);
      document.getElementById('countIncorrect').textContent = toBengaliNumber(wrongCount);

      const percent = (answeredCount / total) * 100;
      document.getElementById('progressBar').style.width = `${percent}%`;
    }

    function setFilter(filter) {
      currentFilter = filter;
      const buttons = {
        all: document.getElementById('btnFilterAll'),
        unanswered: document.getElementById('btnFilterUnanswered'),
        correct: document.getElementById('btnFilterCorrect'),
        incorrect: document.getElementById('btnFilterIncorrect')
      };

      Object.values(buttons).forEach(btn => {
        btn.classList.remove('bg-sky-600', 'text-white', 'shadow-sm');
        if (btn.id === 'btnFilterAll') {
          btn.classList.add('bg-slate-100', 'text-slate-700');
        }
      });

      if (filter === 'all') {
        buttons.all.classList.add('bg-sky-600', 'text-white', 'shadow-sm');
        buttons.all.classList.remove('bg-slate-100', 'text-slate-700');
      } else {
        buttons[filter].classList.add('bg-sky-600', 'text-white', 'shadow-sm');
      }

      renderQuiz();
    }

    function toggleNavModal() {
      const modal = document.getElementById('navModal');
      modal.classList.toggle('hidden');
    }

    function renderPalette() {
      const palette = document.getElementById('paletteGrid');
      palette.innerHTML = '';

      questionsData.forEach(q => {
        const btn = document.createElement('button');
        const isAns = userAnswers[q.id] !== undefined;
        const isCorr = isAns && userAnswers[q.id] === q.correct;

        let style = "py-1.5 rounded-lg text-xs font-bold border transition cursor-pointer ";
        if (!isAns) {
          style += "bg-slate-100 hover:bg-slate-200 text-slate-700 border-slate-300";
        } else if (isCorr) {
          style += "bg-emerald-600 text-white border-emerald-600 shadow-xs";
        } else {
          style += "bg-rose-600 text-white border-rose-600 shadow-xs";
        }

        btn.className = style;
        btn.textContent = toBengaliNumber(q.id);
        btn.onclick = () => {
          toggleNavModal();
          if (currentFilter !== 'all') {
            setFilter('all');
          }
          setTimeout(() => {
            const el = document.getElementById(`q-card-${q.id}`);
            if (el) {
              el.scrollIntoView({ behavior: 'smooth', block: 'center' });
              el.classList.add('ring-2', 'ring-sky-500');
              setTimeout(() => el.classList.remove('ring-2', 'ring-sky-500'), 1600);
            }
          }, 100);
        };
        palette.appendChild(btn);
      });
    }

    function confirmReset() {
      if (!canTakeExam()) {
        showLimitModal();
        return;
      }
      document.getElementById('confirmModal').classList.remove('hidden');
    }

    function closeConfirmModal() {
      document.getElementById('confirmModal').classList.add('hidden');
    }

    function executeReset() {
      closeConfirmModal();
      if (!canTakeExam()) {
        showLimitModal();
        return;
      }
      recordNewAttempt();
      userAnswers = {};
      currentFilter = 'all';
      setFilter('all');
      window.scrollTo({ top: 0, behavior: 'smooth' });
      renderQuiz();
    }

    function showLimitModal() {
      showCountdownForNextAttempt();
      document.getElementById('limitModal').classList.remove('hidden');
    }

    function closeLimitModal() {
      document.getElementById('limitModal').classList.add('hidden');

    }

      }
